  TestWidgetsFlutterBinding.ensureInitialized();
  setUpAll(() async {

    final sora = await File('/tmp/store-Sora-Bold.ttf').readAsBytes();
    fonts.httpClient = MockClient((_) async => http.Response.bytes(sora, 200));
    GoogleFonts.sora(fontWeight: FontWeight.w700);
    await GoogleFonts.pendingFonts();
    final roboto = await File('/Volumes/sontc/Library/flutter/bin/cache/artifacts/material_fonts/Roboto-Regular.ttf').readAsBytes();
    for (final family in ['Ahem', 'Roboto', '-apple-system', 'SF Pro Text']) {
      await (FontLoader(family)..addFont(Future.value(ByteData.sublistView(roboto)))).load();
    }
    final icons = await File('/Volumes/sontc/Library/flutter/bin/cache/artifacts/material_fonts/MaterialIcons-Regular.otf').readAsBytes();
    await (FontLoader('MaterialIcons')..addFont(Future.value(ByteData.sublistView(icons)))).load();
  });
  for (final size in [7, 10]) {
    for (final scenario in ['home', 'txlist', 'report_calendar', 'cats']) {
      testWidgets('tablet $size $scenario', (tester) async {
        SharedPreferences.setMockInitialValues({});
        final prefs = await SharedPreferences.getInstance();
        final logical = size == 7 ? const Size(600, 960) : const Size(800, 1280);
        tester.view.devicePixelRatio = 2;
        tester.view.physicalSize = logical * 2;
        addTearDown(tester.view.resetPhysicalSize);
        addTearDown(tester.view.resetDevicePixelRatio);
        final key = GlobalKey();
        await tester.pumpWidget(RepaintBoundary(key: key, child: _Root(cfgText: '$scenario,vi,0,1.0', prefs: prefs)));
        await tester.pumpAndSettle();
        expect(tester.takeException(), isNull);
        final boundary = key.currentContext!.findRenderObject()! as RenderRepaintBoundary;
        await tester.runAsync(() async {
          final image = await boundary.toImage(pixelRatio: 2);
          final bytes = await image.toByteData(format: ui.ImageByteFormat.png);
          expect(image.width, logical.width * 2);
          expect(image.height, logical.height * 2);
          await File('store_assets/vi-VN/tablet-$size/$scenario.png').writeAsBytes(bytes!.buffer.asUint8List());
          image.dispose();
        });
        await tester.pumpWidget(const SizedBox.shrink());
      });
    }
  }
}
